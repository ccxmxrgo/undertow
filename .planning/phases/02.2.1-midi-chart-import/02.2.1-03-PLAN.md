```gdscript
class_name ChartFile
extends RefCounted
## Single chart-parse choke point: scan and play must parse identically (fingerprint check).
static func is_chart_ext(ext: String) -> bool          # "chart" or "mid" (lowercase compare); ".midi" -> false
static func max_bytes_for(path: String) -> int          # ChartParser.MAX_BYTES for .chart, MidiChartReader.MAX_BYTES for .mid
static func parse(path: String) -> ParsedChart
# - missing / unopenable file -> ok=false, error SongEntry.err_unreadable(0) for .chart, SongEntry.ERR_MIDI_BAD for .mid
# - FileAccess.open + get_length() > max_bytes_for(path) -> ok=false, error SongEntry.ERR_TOO_LARGE (bytes NOT read)
# - else bytes = f.get_buffer(len); "chart" -> ChartParser.parse_bytes; "mid" -> MidiChartReader.parse_bytes
# - any other extension -> ok=false, error SongEntry.err_unreadable(0)
# - parsed.error is always non-empty when ok=false
```
